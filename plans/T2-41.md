class SearchInput(BaseModel):
    schema_version: Literal["2.0"] = "2.0"
    question: str = Field(min_length=1)
    k: int | None = Field(default=None, ge=1, le=20)   # 후보 예산 docs[:k]. None이면 options.answer_k (지금 AskArgs.k와 같은 뜻)

class SearchOutput(BaseModel):
    schema_version: Literal["2.0"] = "2.0"
    question: str
    route: Literal["found", "refuse"]
    refusal_reason: str | None = None          # gate: stage2_reason (out_of_scope | no_item) 또는 None
    refusal_text: str | None = None            # route == refuse 일 때 코드 문구: REFUSALS.get(reason, REFUSAL_TEXT). 호출자가 턴을 끝낼 때 씀. LLM 아님
    picked: list[TableCandidate] = []          # found: rerank2가 골랐으면(stage2_picked, 오류 없음) 그 표들(순위 순), 아니면 candidates. refuse: []
    candidates: list[TableCandidate] = []      # docs[:k] (거절이라도 docs가 있으면 유지 — 지금 refusal_answer와 같음)
    spans: list[dict] = []                     # 노드 순서대로 {"node": name, **trace[name]} (seconds 등)
